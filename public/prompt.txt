        const produtos = [
            {
                id: 1,
                nome: "Alface Lisa Orgânica Maço",
                preco: 7.5,
                img: "./imagem/frutas.jpg"
            },
            {
                id: 2,
                nome: "Tomate",
                preco: 11.9,
                img: ""
            }
        ];

        const money = (v) => v.toLocaleString("pt-BR", {
            style: "currency",
            currency: "BRL"
        });

        const getCart = () => JSON.parse(localStorage.getItem("SaborCart") || "[]");
        const setCart = (p) => localStorage.setItem("SaborCart", JSON.stringify(p));

        /**
         * Função que adiciona um produto ao carrinho
         * @param {*} id :id do produto a ser adicionado
         */
        function add(id) {
            let carrinho = getCart();
            let listaProdutos = produtos.find((x) => x.id === id);
            let indice = carrinho.findIndex((c) => c.id === id);

            if (indice >= 0) {
                carrinho[indice].q++;
            } else {
                carrinho.push({ ...listaProdutos, q: 1 });
            }

            setCart(carrinho);
            toast(`${listaProdutos.nome} adicionando ao carrinho!`);
        }

        function toast(texto) {
            const elemento = document.getElementById("toast");
            elemento.textContent = texto;
            elemento.classList.add("show");

            setTimeout(() => elemento.classList.remove("show"), 1000);
        }

        function render(list = produtos) {
            document.getElementById("productGrid").innerHTML = list.map((produto) =>
                `<article class="card">
        <button class="heart">favoritar</button>
        <img src="${produto.img}" alt="${produto.nome}">
        <h3>${produto.nome}</h3>
        <div class="price">${money(produto.preco)}</div>
        <button class="add" onClick="add(${produto.id})">Adicionar</button>
        </article>`
            ).join("");
        }

        function search() {
            const elemento = document.getElementById("searchInput");
            const texto = elemento.value.toLowrCase();

            render(
                produtos.filter((produto) =>
                    produto.nome.toLoweCase().includes(texto)
                )
            );
        }

        document.getElementById("searchForm").addEventListener(
            "submit",
            (evento) => {
                evento.preventDefault();
                search();
            }
        );

        document.getElementById("searchInput").addEventListener("input", search);

        document.getAnimations.getElementById("sort").addEventListener(
            "change",
            (evento) => {
                let lista = [...produtos];

                if (evento.target.value === "az") {
                    lista.sort((a, b) => a.nome.localeCompare(b.nome));
                } else {
                    lista.sort((a, b) => a.preco - b.preco);
                }

                render(lista);
            }
        );


    </script>
